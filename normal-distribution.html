function draw(){ const W = canvas.clientWidth; const H = canvas.clientHeight; ctx.clearRect(0,0,W,H); const mu = parseFloat(meanInput.value); const sigma = parseFloat(sdInput.value); if(!Number.isFinite(mu)) return; if(!Number.isFinite(sigma) || sigma <= 0) return; const variance = sigma * sigma; distribution.innerHTML = "X ~ N(" + formatNumber(mu) + ", " + formatNumber(variance) + ")"; const L = 55; const R = 25; const T = 35; const B = 48; const gw = W - L - R; const gh = H - T - B; /* ===================================================== FIXED X-AXIS THIS IS WHAT MAKES THE BELL ACTUALLY MOVE. Changing μ moves the bell LEFT or RIGHT. Changing σ changes the WIDTH and HEIGHT. ===================================================== */ const xmin = 100; const xmax = 240; /* FIXED VERTICAL SCALE This allows students to see σ change the height as well as the width. */ const referenceSigma = 5; const ymax = pdf(170,170,referenceSigma) * 1.15; const xp = x => L + (x-xmin)/(xmax-xmin)*gw; const yp = y => T + gh - (y/ymax)*gh; /* GRID */ ctx.strokeStyle = "#f0f1f3"; ctx.lineWidth = 1; ctx.fillStyle = "#667085"; ctx.font = "11px Arial"; ctx.textAlign = "center"; for(let x=100; x<=240; x+=20){ const px = xp(x); ctx.beginPath(); ctx.moveTo(px,T); ctx.lineTo(px,T+gh); ctx.stroke(); ctx.fillText( x, px, T+gh+21 ); } /* X-AXIS */ ctx.strokeStyle = "#667085"; ctx.lineWidth = 1.2; ctx.beginPath(); ctx.moveTo(L,T+gh); ctx.lineTo(L+gw,T+gh); ctx.stroke(); /* SHADED AREA */ ctx.beginPath(); ctx.moveTo(xp(xmin),yp(0)); const samples = 800; for(let i=0;i<=samples;i++){ const x = xmin + (i/samples)*(xmax-xmin); ctx.lineTo( xp(x), yp(pdf(x,mu,sigma)) ); } ctx.lineTo( xp(xmax), yp(0) ); ctx.closePath(); ctx.fillStyle = "rgba(156,20,88,.09)"; ctx.fill(); /* BELL CURVE */ ctx.beginPath(); for(let i=0;i<=samples;i++){ const x = xmin + (i/samples)*(xmax-xmin); const X = xp(x); const Y = yp(pdf(x,mu,sigma)); if(i===0) ctx.moveTo(X,Y); else ctx.lineTo(X,Y); } ctx.strokeStyle = "#9c1458"; ctx.lineWidth = 3; ctx.stroke(); /* MEAN LINE */ if(mu >= xmin && mu <= xmax){ const mx = xp(mu); const peakY = yp(pdf(mu,mu,sigma)); ctx.save(); ctx.setLineDash([5,5]); ctx.strokeStyle = "#1565c0"; ctx.lineWidth = 2; ctx.beginPath(); ctx.moveTo(mx,T+gh); ctx.lineTo(mx,peakY); ctx.stroke(); ctx.restore(); /* μ LABEL */ ctx.fillStyle = "#1565c0"; ctx.font = "bold 15px Arial"; ctx.fillText( "μ", mx, T+gh+39 ); } /* μ − σ AND μ + σ */ ctx.strokeStyle = "#9c1458"; ctx.lineWidth = 2; [mu-sigma,mu+sigma].forEach(x=>{ if(x >= xmin && x <= xmax){ const px = xp(x); ctx.beginPath(); ctx.moveTo(px,T+gh-5); ctx.lineTo(px,T+gh+5); ctx.stroke(); } }); ctx.fillStyle = "#9c1458"; ctx.font = "bold 11px Arial"; if(mu-sigma >= xmin && mu-sigma <= xmax){ ctx.fillText( "μ − σ", xp(mu-sigma), T+gh-11 ); } if(mu+sigma >= xmin && mu+sigma <= xmax){ ctx.fillText( "μ + σ", xp(mu+sigma), T+gh-11 ); } }
